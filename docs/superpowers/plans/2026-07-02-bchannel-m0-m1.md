# bChannel M0+M1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bootstrap the `bChannel` monorepo (M0) and ship the M1 platform: a NIP-29 relay, browser identity + key vault, NIP-44-encrypted channels with invite/join and key rotation, NIP-17 DMs, and a one-command docker compose stack.

**Architecture:** Event-sourced on a NIP-29 relay (relay29/khatru29). Every channel's content is NIP-44-encrypted to a per-channel epoch key distributed via NIP-59 gift wrap and rotated on member removal. Identity keys live in a browser key vault (password-scrypt-wrapped, passkey-PRF where available). dkgkit compiles to WASM and is proven by native↔WASM↔noble parity tests (ceremonies use it in M2).

**Tech Stack:** Bun workspaces, TypeScript strict, vitest 4, Biome, React 19 + Vite 7 (PWA), nostr-tools 2.23, @noble/curves+hashes+ciphers, @scure/base, relay29/khatru29 (Go) + badger eventstore, wasm-bindgen + dkgkit (git), Playwright.

`crates/dkgkit-wasm/src/lib.rs`:

```rust
//! Minimal WASM surface over dkgkit HTSS: enough to prove the toolchain and
//! the math end-to-end (M0). M2 adds round-by-round bindings for real
//! multi-device ceremonies.

use dkgkit_core::{HierarchicalThresholdConfig, ParticipantId, RankedParticipant};
use dkgkit_frost::{run_distributed_htss_keygen, sign_digest_with_local_htss_threshold_shares};
use wasm_bindgen::prelude::*;

/// Runs a full flat-rank HTSS DKG + threshold sign locally.
/// Returns (group xonly pubkey hex, 64-byte BIP340 signature hex).
pub fn ceremony_demo_core(
    threshold: u16,
    participants: u16,
    digest: [u8; 32],
) -> Result<(String, String), String> {
    let ranked = (1..=participants)
        .map(|i| RankedParticipant::new(i, 0, None).map_err(|e| e.to_string()))
        .collect::<Result<Vec<_>, _>>()?;
    let config =
        HierarchicalThresholdConfig::new(threshold, ranked).map_err(|e| e.to_string())?;
    let keyset = run_distributed_htss_keygen(&config).map_err(|e| e.to_string())?;
    let signer_set: Vec<ParticipantId> = keyset
        .shares
        .iter()
        .take(threshold as usize)
        .map(|s| s.participant_id)
        .collect();
    let sig = sign_digest_with_local_htss_threshold_shares(
        &keyset.group_key,
        digest,
        &keyset.shares,
        &signer_set,
        &config,
    )
    .map_err(|e| e.to_string())?;
    Ok((
        hex::encode(keyset.group_key.xonly_public_key),
        hex::encode(&sig.signature_bytes),
    ))
}

#[wasm_bindgen]
pub fn ceremony_demo(
    threshold: u16,
    participants: u16,
    digest_hex: &str,
) -> Result<String, JsValue> {
    let digest_vec = hex::decode(digest_hex).map_err(|e| JsValue::from_str(&e.to_string()))?;
    let digest: [u8; 32] = digest_vec
        .try_into()
        .map_err(|_| JsValue::from_str("digest must be exactly 32 bytes"))?;
    let (xonly, signature) =
        ceremony_demo_core(threshold, participants, digest).map_err(|e| JsValue::from_str(&e))?;
    Ok(serde_json::json!({ "xonly": xonly, "signature": signature }).to_string())
}
```

```ts
import type { Event as NostrEvent } from "nostr-tools";
import { decrypt, encrypt } from "nostr-tools/nip44";
import { unwrapEvent, wrapEvent } from "nostr-tools/nip59";
import { KIND_CHANNEL_KEY } from "./kinds";

export interface ChannelKeyEnvelope {
  channelId: string;
  epoch: number;
  key: Uint8Array;
}

export function generateChannelKey(): Uint8Array {
  const key = new Uint8Array(32);
  globalThis.crypto.getRandomValues(key);
  return key;
}

const toHex = (b: Uint8Array) =>
  Array.from(b, (x) => x.toString(16).padStart(2, "0")).join("");
const fromHex = (h: string) =>
  new Uint8Array(h.match(/.{2}/g)?.map((x) => Number.parseInt(x, 16)) ?? []);

/** Gift-wrap (NIP-59) a channel epoch key to one member. */
export function sealChannelKey(
  env: ChannelKeyEnvelope,
  senderSecret: Uint8Array,
  recipientPub: string,
): NostrEvent {
  return wrapEvent(
    {
      kind: KIND_CHANNEL_KEY,
      content: JSON.stringify({ channelId: env.channelId, epoch: env.epoch, key: toHex(env.key) }),
      tags: [["h", env.channelId]],
    },
    senderSecret,
    recipientPub,
  );
}

/** Returns null when the wrap is not for us or not a channel-key envelope. */
export function openChannelKey(
  wrap: NostrEvent,
  recipientSecret: Uint8Array,
): ChannelKeyEnvelope | null {
  try {
    const rumor = unwrapEvent(wrap, recipientSecret);
    if (rumor.kind !== KIND_CHANNEL_KEY) return null;
    const body = JSON.parse(rumor.content) as { channelId: string; epoch: number; key: string };
    if (typeof body.channelId !== "string" || typeof body.epoch !== "number") return null;
    const key = fromHex(body.key);
    if (key.length !== 32) return null;
    return { channelId: body.channelId, epoch: body.epoch, key };
  } catch {
    return null;
  }
}

/** NIP-44 with a raw 32-byte symmetric channel key as the conversation key. */
export function encryptChannelMessage(key: Uint8Array, plaintext: string): string {
  return encrypt(plaintext, key);
}

export function decryptChannelMessage(key: Uint8Array, payload: string): string {
  return decrypt(payload, key);
}
```

```ts
import { decryptChannelMessage, encryptChannelMessage } from "./channel-key";

export interface ChannelKeyring {
  channelId: string;
  epochs: Map<number, Uint8Array>;
}

export function createKeyring(channelId: string): ChannelKeyring {
  return { channelId, epochs: new Map() };
}

export function addEpochKey(kr: ChannelKeyring, epoch: number, key: Uint8Array): void {
  kr.epochs.set(epoch, key);
}

export function latestEpoch(kr: ChannelKeyring): number {
  return kr.epochs.size === 0 ? -1 : Math.max(...kr.epochs.keys());
}

export function encryptLatest(
  kr: ChannelKeyring,
  plaintext: string,
): { content: string; epoch: number } {
  const epoch = latestEpoch(kr);
  const key = kr.epochs.get(epoch);
  if (key == null) throw new Error(`keyring for ${kr.channelId} has no keys`);
  return { content: encryptChannelMessage(key, plaintext), epoch };
}

export function decryptAtEpoch(kr: ChannelKeyring, epoch: number, payload: string): string {
  const key = kr.epochs.get(epoch);
  if (key == null) throw new Error(`unknown epoch ${epoch} for channel ${kr.channelId}`);
  return decryptChannelMessage(key, payload);
}
```

```ts
import { xchacha20poly1305 } from "@noble/ciphers/chacha.js";
import { scrypt } from "@noble/hashes/scrypt.js";
import { fromHex, randomBytes, toHex, type KeyWrap, type WrappedSecret } from "./keywrap";

const KDF = { N: 2 ** 17, r: 8, p: 1, dkLen: 32 };

export class PasswordKeyWrap implements KeyWrap {
  constructor(private readonly password: string) {}

  async wrap(secret: Uint8Array): Promise<WrappedSecret> {
    const salt = randomBytes(16);
    const nonce = randomBytes(24);
    const key = scrypt(this.password, salt, KDF);
    const ciphertext = xchacha20poly1305(key, nonce).encrypt(secret);
    return {
      alg: "password-scrypt-xchacha",
      saltHex: toHex(salt),
      nonceHex: toHex(nonce),
```

```ts
/** WebAuthn PRF wrap. The PRF output for a fixed salt is a stable 32-byte
 * secret bound to the platform authenticator; we use it as the wrapping key.
 * Availability is feature-detected; Login falls back to PasswordKeyWrap. */
const PRF_SALT = new TextEncoder().encode("bchannel-vault-v1");

export async function isPasskeySupported(): Promise<boolean> {
  return (
    typeof PublicKeyCredential !== "undefined" &&
    (await PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable?.()) === true
  );
}

async function prfKey(credentialId?: Uint8Array): Promise<{ key: Uint8Array; id: Uint8Array }> {
  const assertion = (await navigator.credentials.get({
    publicKey: {
      challenge: randomBytes(32),
      userVerification: "required",
      allowCredentials: credentialId ? [{ id: credentialId, type: "public-key" }] : [],
      extensions: { prf: { eval: { first: PRF_SALT } } },
    },
  })) as PublicKeyCredential;
  const prf = assertion.getClientExtensionResults().prf?.results?.first;
  if (!prf) throw new Error("authenticator does not support the PRF extension");
  return { key: new Uint8Array(prf as ArrayBuffer), id: new Uint8Array(assertion.rawId) };
}

export async function createPasskey(userName: string): Promise<Uint8Array> {
  const cred = (await navigator.credentials.create({
    publicKey: {
      challenge: randomBytes(32),
      rp: { name: "bChannel" },
      user: { id: randomBytes(16), name: userName, displayName: userName },
      pubKeyCredParams: [{ alg: -7, type: "public-key" }, { alg: -257, type: "public-key" }],
      authenticatorSelection: { residentKey: "required", userVerification: "required" },
      extensions: { prf: {} },
    },
  })) as PublicKeyCredential;
  return new Uint8Array(cred.rawId);
}

export class PasskeyKeyWrap implements KeyWrap {
  constructor(private readonly credentialId?: Uint8Array) {}

  async wrap(secret: Uint8Array): Promise<WrappedSecret> {
    const { key, id } = await prfKey(this.credentialId);
    const nonce = randomBytes(24);
    return {
      alg: "passkey-prf-xchacha",
      saltHex: toHex(PRF_SALT),
      nonceHex: toHex(nonce),
      ciphertextHex: toHex(xchacha20poly1305(key, nonce).encrypt(secret)),
      credentialIdB64: btoa(String.fromCharCode(...id)),
    };
  }

  async unwrap(w: WrappedSecret): Promise<Uint8Array> {
    const credId = w.credentialIdB64
      ? new Uint8Array([...atob(w.credentialIdB64)].map((c) => c.charCodeAt(0)))
      : undefined;
    const { key } = await prfKey(credId);
    return xchacha20poly1305(key, fromHex(w.nonceHex)).decrypt(fromHex(w.ciphertextHex));
  }
}
```

```ts
import { finalizeEvent, type Event, type Filter } from "nostr-tools";
import { Relay } from "nostr-tools/relay";

/** One relay connection with lazy connect and NIP-42 auth. */
export class NostrClient {
  private relay: Relay | null = null;
  private authSk: Uint8Array | null = null;

  constructor(private readonly url: string) {}

  setAuthSecret(sk: Uint8Array): void {
    this.authSk = sk;
  }

  private async conn(): Promise<Relay> {
    if (this.relay?.connected) return this.relay;
    this.relay = await Relay.connect(this.url);
    this.relay.onauth = async (evt) => {
      if (!this.authSk) throw new Error("relay demanded AUTH but no secret is set");
      return finalizeEvent(evt, this.authSk);
    };
    return this.relay;
  }

  async publish(evt: Event): Promise<void> {
    const r = await this.conn();
    await r.publish(evt);
  }

  async list(filters: Filter[]): Promise<Event[]> {
    const r = await this.conn();
    const out: Event[] = [];
    await new Promise<void>((resolve) => {
      const sub = r.subscribe(filters, {
        onevent: (e) => out.push(e),
        oneose: () => {
          sub.close();
          resolve();
        },
      });
    });
    return out;
  }

  subscribe(filters: Filter[], onevent: (e: Event) => void): () => void {
    let closed = false;
    let close = () => {
      closed = true;
    };
    this.conn().then((r) => {
      if (closed) return;
      const sub = r.subscribe(filters, { onevent });
      close = () => sub.close();
    });
    return () => close();
  }

  close(): void {
    this.relay?.close();
    this.relay = null;
  }
}

export const RELAY_URL: string =
  (import.meta as { env?: Record<string, string> }).env?.VITE_RELAY_URL ?? "ws://localhost:5577";
```

```ts
import { NIP29 } from "@bchannel/protocol";
import { finalizeEvent, type Event } from "nostr-tools";
import type { NostrClient } from "./client";

const now = () => Math.floor(Date.now() / 1000);
const sign = (sk: Uint8Array, kind: number, tags: string[][], content = "") =>
  finalizeEvent({ kind, tags, content, created_at: now() }, sk);

export const createGroupEvent = (sk: Uint8Array, groupId: string): Event =>
  sign(sk, NIP29.CREATE_GROUP, [["h", groupId]]);

export const editMetadataEvent = (
  sk: Uint8Array,
  groupId: string,
  meta: { name?: string; about?: string; priv?: boolean; closed?: boolean },
): Event => {
  const tags: string[][] = [["h", groupId]];
  if (meta.name) tags.push(["name", meta.name]);
  if (meta.about) tags.push(["about", meta.about]);
  if (meta.priv) tags.push(["private"]);
  if (meta.closed) tags.push(["closed"]);
  return sign(sk, NIP29.EDIT_METADATA, tags);
};

export const putUserEvent = (sk: Uint8Array, groupId: string, memberPub: string): Event =>
  sign(sk, NIP29.PUT_USER, [["h", groupId], ["p", memberPub]]);

export const removeUserEvent = (sk: Uint8Array, groupId: string, memberPub: string): Event =>
  sign(sk, NIP29.REMOVE_USER, [["h", groupId], ["p", memberPub]]);

export const createInviteEvent = (sk: Uint8Array, groupId: string, code: string): Event =>
  sign(sk, NIP29.CREATE_INVITE, [["h", groupId], ["code", code]]);

export const joinRequestEvent = (sk: Uint8Array, groupId: string, code?: string): Event =>
  sign(sk, NIP29.JOIN_REQUEST, code ? [["h", groupId], ["code", code]] : [["h", groupId]]);

export const leaveRequestEvent = (sk: Uint8Array, groupId: string): Event =>
  sign(sk, NIP29.LEAVE_REQUEST, [["h", groupId]]);

export async function fetchGroupMembers(client: NostrClient, groupId: string): Promise<string[]> {
  const evts = await client.list([{ kinds: [NIP29.GROUP_MEMBERS], "#d": [groupId] }]);
  const latest = evts.sort((a, b) => b.created_at - a.created_at)[0];
  return latest?.tags.filter((t) => t[0] === "p").map((t) => t[1] as string) ?? [];
}

export async function fetchMyGroupIds(client: NostrClient, pubkey: string): Promise<string[]> {
  const evts = await client.list([{ kinds: [NIP29.GROUP_MEMBERS], "#p": [pubkey] }]);
  return [...new Set(evts.flatMap((e) => e.tags.filter((t) => t[0] === "d").map((t) => t[1] as string)))];
}

export async function fetchGroupName(client: NostrClient, groupId: string): Promise<string | null> {
  const evts = await client.list([{ kinds: [NIP29.GROUP_METADATA], "#d": [groupId] }]);
  const latest = evts.sort((a, b) => b.created_at - a.created_at)[0];
  return latest?.tags.find((t) => t[0] === "name")?.[1] ?? null;
}
```

```ts
export function membersNeedingKey(members: string[], sealed: Set<string>): string[] {
  return members.filter((m) => !sealed.has(m));

export function buildInviteHash(groupId: string, code: string, relay: string): string {
  const q = new URLSearchParams({ group: groupId, code, relay });

  const code = q.get("code");
```

```ts
import type { Identity } from "../vault/vault";
import type { NostrClient } from "./client";

export type Rumor = ReturnType<typeof unwrapEvent>;

export function startInbox(
  client: NostrClient,
  identity: Identity,
  handlers: {
    onChannelKey: (env: ChannelKeyEnvelope) => void;
    onDmRumor: (rumor: Rumor) => void;
  },
): () => void {
  return client.subscribe([{ kinds: [KIND_GIFT_WRAP], "#p": [identity.pubkey] }], (wrap) => {
    const env = openChannelKey(wrap, identity.secretKey);
    if (env) {
      handlers.onChannelKey(env);
      return;
    }
    try {
      const rumor = unwrapEvent(wrap, identity.secretKey);
      if (rumor.kind === KIND_DM_RUMOR) handlers.onDmRumor(rumor);
      if (rumor.kind === KIND_CHANNEL_KEY) return; // already handled above
    } catch {
      /* not for us */
    }
  });
}
```

```tsx
  const [groupIds, setGroupIds] = useState<string[]>([]);

  const sealedRef = useRef<Map<string, Set<string>>>(new Map());

  useEffect(() => {
    vaultExists().then(setHasVault);
  }, []);

  // connect + inbox once logged in
  useEffect(() => {
    if (!identity) return;
    const c = new NostrClient(invite?.relay ?? RELAY_URL);
    c.setAuthSecret(identity.secretKey);
    setClient(c);
    const stopInbox = startInbox(c, identity, {
      onChannelKey: (env) =>
        setKeyrings((prev) => {
          const next = new Map(prev);
          const kr = next.get(env.channelId) ?? createKeyring(env.channelId);
          addEpochKey(kr, env.epoch, env.key);
          next.set(env.channelId, kr);
          return next;
        }),
      onDmRumor: () => {}, // wired to the DM store in Task 9
    });
    // spec §2: profile names come from kind-0 metadata on the company relay
    const displayName = localStorage.getItem("bchannel:displayName");
    if (displayName) {
      import("nostr-tools").then(({ finalizeEvent }) =>
        c.publish(
          finalizeEvent(
            {
              kind: 0,
              content: JSON.stringify({ name: displayName }),
              tags: [],
              created_at: Math.floor(Date.now() / 1000),
            },
            identity.secretKey,
          ),
        ),
      );
    }
    return () => {
      stopInbox();
      c.close();
    };
  }, [identity]);

    if (!client || !identity) return;
    setGroupIds(await fetchMyGroupIds(client, identity.pubkey));
  }, [client, identity]);

  useEffect(() => {
    refreshGroups();
    const t = setInterval(refreshGroups, 3000);
    return () => clearInterval(t);
  }, [refreshGroups]);

  // admin duty: seal current key to new members; rotate on removals

    if (!client || !identity) return;

      for (const [gid, kr] of keyrings) {
        // creator-only duty: sealedRef is seeded only in onCreated. Without
        // this gate every key-holder would race to rotate/seal concurrently.
        if (!sealedRef.current.has(gid)) continue;
        const current = await fetchGroupMembers(client, gid);
        const prev = membersRef.current.get(gid) ?? current;
        membersRef.current.set(gid, current);
        const epoch = latestEpoch(kr);
        if (epoch < 0) continue;

        if (removedMembers(prev, current).length > 0) {
          const newKey = generateChannelKey();

              sealChannelKey({ channelId: gid, epoch: newEpoch, key: newKey }, identity.secretKey, m),
            );
            sealedRef.current.get(gid)?.add(m);
          }
          continue;
        }

        const sealed = sealedRef.current.get(gid) ?? new Set<string>();
        const key = kr.epochs.get(epoch);
        if (!key) continue;
        for (const m of membersNeedingKey(current, sealed)) {
          await client.publish(
            sealChannelKey({ channelId: gid, epoch, key }, identity.secretKey, m),
          );
          sealed.add(m);
        }
        sealedRef.current.set(gid, sealed);
      }
    }, 2000);
    return () => clearInterval(t);
  }, [client, identity, keyrings]);

  if (!identity) {
    if (hasVault == null) return <p>loading…</p>;
    return <Login hasVault={hasVault} onUnlock={setIdentity} />;
  }
  if (!client) return <p>connecting…</p>;
  if (invite) {
    return (
      <InviteJoin
        client={client}
        identity={identity}
        invite={invite}
        onJoined={() => {
          window.location.hash = "";
          refreshGroups();
        }}
      />
    );
  }
  return (
    <div style={{ display: "flex", gap: 16 }}>
      <ChannelList
        client={client}
        identity={identity}
        groupIds={groupIds}
        active={active}
        onSelect={setActive}

          const key = generateChannelKey();
          addEpochKey(kr, 0, key);
          setKeyrings((prev) => new Map(prev).set(gid, kr));
          client.publish(
            sealChannelKey({ channelId: gid, epoch: 0, key }, identity.secretKey, identity.pubkey),
          );
          sealedRef.current.set(gid, new Set([identity.pubkey]));
          refreshGroups();
          setActive(gid);
        }}

        {active ? (
          <ChannelView
            client={client}
            identity={identity}
            groupId={active}
            keyring={keyrings.get(active) ?? createKeyring(active)}

        ) : (
          <p>select a channel</p>
        )}
        {/* DMs UI (uses dmRumors) is added by Task 9 */}
      </div>
    </div>
```

```tsx
import {
  KIND_CHAT,
  decryptAtEpoch,
  encryptLatest,
  latestEpoch,
  type ChannelKeyring,
} from "@bchannel/protocol";
import { finalizeEvent, type Event } from "nostr-tools";
import { useEffect, useState } from "react";
import type { NostrClient } from "../nostr/client";
import type { Identity } from "../vault/vault";

export default function ChannelView(props: {
  client: NostrClient;
  identity: Identity;
  groupId: string;
  keyring: ChannelKeyring;
}) {
  const [events, setEvents] = useState<Event[]>([]);
  const [draft, setDraft] = useState("");
  const ready = latestEpoch(props.keyring) >= 0;

  useEffect(() => {
    setEvents([]);
    return props.client.subscribe([{ kinds: [KIND_CHAT], "#h": [props.groupId] }], (e) =>
      setEvents((prev) => (prev.some((p) => p.id === e.id) ? prev : [...prev, e])),
    );
  }, [props.client, props.groupId]);

  function render(e: Event): string {
    const ke = Number(e.tags.find((t) => t[0] === "ke")?.[1]);
    try {
      return decryptAtEpoch(props.keyring, ke, e.content);
    } catch {
      return "🔒 (no key for this epoch yet)";
    }
  }

  async function send() {
    const { content, epoch } = encryptLatest(props.keyring, draft);
    await props.client.publish(
      finalizeEvent(
        {
          kind: KIND_CHAT,
          content,
          tags: [["h", props.groupId], ["ke", String(epoch)]],
          created_at: Math.floor(Date.now() / 1000),
        },
        props.identity.secretKey,
      ),
    );
    setDraft("");
  }

  return (
    <section>
      <h2>#{props.groupId}</h2>
      {!ready && <p>waiting for the channel key…</p>}
      <ul data-testid="message-list">
        {events
          .sort((a, b) => a.created_at - b.created_at)
          .map((e) => (
            <li key={e.id}>
              <code>{e.pubkey.slice(0, 8)}</code>: {render(e)}
            </li>
          ))}
      </ul>
      <input
        data-testid="message-input"
        value={draft}
        disabled={!ready}
        onChange={(e) => setDraft(e.target.value)}
        onKeyDown={(e) => e.key === "Enter" && draft && send()}
      />
      <button type="button" data-testid="message-send" disabled={!ready || !draft} onClick={send}>
        Send
      </button>
    </section>
  );
}
```

```ts
import { wrapManyEvents } from "nostr-tools/nip17";
import type { Rumor } from "./inbox";
import type { NostrClient } from "./client";

/** Wraps to the recipient and to self (sender's own readable copy). */
export async function sendDm(
  client: NostrClient,
  sk: Uint8Array,
  myPub: string,
  recipientPub: string,
  text: string,
): Promise<void> {
  const wraps = wrapManyEvents(sk, [{ publicKey: recipientPub }, { publicKey: myPub }], text);
  for (const w of wraps) await client.publish(w);
}

export function dmPartner(rumor: Rumor, myPub: string): string {
  if (rumor.pubkey !== myPub) return rumor.pubkey;
  return rumor.tags.find((t) => t[0] === "p")?.[1] ?? rumor.pubkey;
}
```

```ts
  const admin = await adminCtx.newPage();
  const member = await memberCtx.newPage();

  // admin: create identity + channel
  await admin.goto("/");
  await admin.getByTestId("pw-input").fill("admin-pass");
  await admin.getByTestId("login-submit").click();
  await admin.getByTestId("new-channel-name").fill("treasury");
  await admin.getByTestId("new-channel-create").click();
  const inviteLink = await admin.getByTestId("invite-link").textContent();
  expect(inviteLink).toBeTruthy();

  // member: join via invite link
  await member.goto(inviteLink as string);
  await member.getByTestId("pw-input").fill("member-pass");
  await member.getByTestId("login-submit").click();
  await expect(member.getByTestId("join-status")).toHaveText("joined!", { timeout: 15_000 });

  // member opens the channel and waits for the gift-wrapped key (admin page is
  // open, so its poller seals the current epoch key to the new member)
  await member.getByTestId(/channel-item-/).first().click();
  await expect(member.getByTestId("message-input")).toBeEnabled({ timeout: 20_000 });

  // member sends; admin reads
  await member.getByTestId("message-input").fill("hello from member");
  await member.getByTestId("message-send").click();
  await admin.getByTestId(/channel-item-/).first().click();
  await expect(admin.getByTestId("message-list")).toContainText("hello from member", {
    timeout: 15_000,
  });
});
```
